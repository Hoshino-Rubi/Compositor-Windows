cmake_minimum_required(VERSION 3.24)
project(CompositorWindows VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Concurrent Test Svg)
set(COMPOSITOR_DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../.tools/vcpkg/installed/x64-windows" CACHE PATH "Native decoder dependencies")
list(APPEND CMAKE_PREFIX_PATH "${COMPOSITOR_DEPS}")
find_package(libraw CONFIG REQUIRED)
find_package(libheif CONFIG REQUIRED)
find_package(ZLIB REQUIRED)
set(COMPOSITOR_ONNX "${CMAKE_CURRENT_SOURCE_DIR}/../.tools/onnxruntime/package" CACHE PATH "ONNX Runtime SDK")
add_library(onnxruntime SHARED IMPORTED)
set_target_properties(onnxruntime PROPERTIES IMPORTED_IMPLIB "${COMPOSITOR_ONNX}/runtimes/win-x64/native/onnxruntime.lib" IMPORTED_LOCATION "${COMPOSITOR_ONNX}/runtimes/win-x64/native/onnxruntime.dll" INTERFACE_INCLUDE_DIRECTORIES "${COMPOSITOR_ONNX}/build/native/include")
if(MSVC)
  add_compile_options(/utf-8 /W4 /permissive-)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(upstream_pixels STATIC ../Compositor/Rendering/AdjustPixels.c ../Compositor/Rendering/LensPixels.c ../Compositor/Rendering/NoisePixels.c ../Compositor/Rendering/LevelsPixels.c ../Compositor/Rendering/HealPixels.c ../Compositor/Rendering/ContentFill.c)
target_include_directories(upstream_pixels PUBLIC ../Compositor/Rendering)
if(MSVC)
  target_compile_definitions(upstream_pixels PRIVATE _USE_MATH_DEFINES)
endif()
add_library(compositor_core STATIC src/Project.cpp src/Project.h src/Renderer.cpp src/Renderer.h src/Document.cpp src/Document.h src/DocumentOperations.cpp src/Filters.cpp src/Filters.h src/Effects.cpp src/Effects.h src/Localization.cpp src/Localization.h src/Photoshop.cpp src/ImageDecoders.cpp src/Importers.h src/CameraRaw.cpp src/CameraRaw.h src/Warp.cpp src/Warp.h src/Selection.cpp src/Selection.h src/Text.cpp src/Text.h)
target_include_directories(compositor_core PUBLIC src)
target_compile_definitions(compositor_core PRIVATE COMPOSITOR_MODEL_PATH="${CMAKE_CURRENT_SOURCE_DIR}/resources/models/u2netp.onnx")
target_link_libraries(compositor_core PUBLIC Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Svg upstream_pixels libraw::raw_r heif ZLIB::ZLIB onnxruntime)
qt_add_resources(compositor_core translations PREFIX "/" FILES translations/en.json translations/zh-CN.json translations/zh-TW.json)
add_library(compositor_ui STATIC src/Canvas.cpp src/Canvas.h src/Icons.cpp src/Icons.h src/WorkspaceUi.cpp src/WorkspaceActions.cpp src/FilterDialog.cpp src/TextDialog.cpp src/EffectsDialog.cpp src/CameraRawDialog.cpp src/ImportDialog.cpp src/WorkspaceAdvanced.cpp src/MainWindow.cpp src/MainWindow.h)
target_link_libraries(compositor_ui PUBLIC compositor_core Qt6::Concurrent)
qt_add_resources(compositor_ui icons PREFIX "/icons" BASE "../Compositor/Assets.xcassets/AppIcon.appiconset" FILES "../Compositor/Assets.xcassets/AppIcon.appiconset/app-icon-128.png")
qt_add_resources(compositor_ui controls PREFIX "/controls" BASE "resources" FILES resources/chevron-down.svg)
qt_add_executable(Compositor WIN32 src/main.cpp)
if(WIN32)
  target_sources(Compositor PRIVATE resources/app.rc)
  target_include_directories(Compositor PRIVATE resources)
endif()
target_link_libraries(Compositor PRIVATE compositor_ui)
include(CTest)
if(BUILD_TESTING)
  qt_add_executable(compositor_tests tests/Tests.cpp)
  target_link_libraries(compositor_tests PRIVATE compositor_ui Qt6::Test)
  target_compile_definitions(compositor_tests PRIVATE COMPOSITOR_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  add_test(NAME compositor_native COMMAND compositor_tests -platform offscreen -o native-tests.xml,xml)
  set_tests_properties(compositor_native PROPERTIES TIMEOUT 120)
endif()
install(TARGETS Compositor RUNTIME DESTINATION .)
install(FILES ../LICENSE DESTINATION licenses RENAME Compositor-MIT.txt)




# Keep ONNX Runtime beside the executable; Windows also ships a different DLL with the same name.
foreach(app Compositor compositor_tests)
  if(TARGET ${app})
    add_custom_command(TARGET ${app} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${COMPOSITOR_ONNX}/runtimes/win-x64/native/onnxruntime.dll" "$<TARGET_FILE_DIR:${app}>/onnxruntime.dll" COMMAND ${CMAKE_COMMAND} -E copy_if_different "${COMPOSITOR_ONNX}/runtimes/win-x64/native/onnxruntime_providers_shared.dll" "$<TARGET_FILE_DIR:${app}>/onnxruntime_providers_shared.dll")
  endif()
endforeach()

set(COMPOSITOR_NATIVE_DLLS raw_r.dll jasper.dll jpeg62.dll lcms2-2.dll heif.dll libde265.dll z.dll)
foreach(app Compositor compositor_tests)
  if(TARGET ${app})
    foreach(dll IN LISTS COMPOSITOR_NATIVE_DLLS)
      add_custom_command(TARGET ${app} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${COMPOSITOR_DEPS}/bin/${dll}" "$<TARGET_FILE_DIR:${app}>/${dll}")
    endforeach()
  endif()
endforeach()
install(DIRECTORY resources/models DESTINATION .)
