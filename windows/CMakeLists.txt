# Generated wide paths are checked in so users do not need Python to build.
function(compositor_check_wide original generated)
  file(READ "${original}" input)
  string(REPLACE "\r\n" "\n" input "${input}")
  string(SHA256 digest "${input}")
  file(READ "${generated}" output)
  if(NOT output MATCHES "Source SHA256: ${digest}")
    message(FATAL_ERROR "Stale 16-bit implementation: ${generated}. Regenerate with windows/scripts/generate-wide-*.py")
  endif()
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${original}" "${generated}")
endfunction()
foreach(kernel AdjustPixels LensPixels NoisePixels DitherPixels HealPixels ContentFill)
  compositor_check_wide("${CMAKE_CURRENT_SOURCE_DIR}/../Compositor/Rendering/${kernel}.c" "${CMAKE_CURRENT_SOURCE_DIR}/src/Wide${kernel}.cpp")
endforeach()
foreach(kernel Renderer Filters Effects CameraRaw Warp)
  compositor_check_wide("${CMAKE_CURRENT_SOURCE_DIR}/src/${kernel}.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/src/Wide${kernel}.cpp")
endforeach()
